    allowed = { always = yes }
    enable = {
        is_ai = yes
        pwc_st_is_city_country = yes

            }
        }

        type = unit_ratio

        value = 0
    }

        type = equipment_variant_production_factor
        id = pwc_platform_cv_cas
        value = -100
    }
}
