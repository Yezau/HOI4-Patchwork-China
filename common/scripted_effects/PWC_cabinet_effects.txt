pwc_se_appoint_seat_civil_admin_1 = {
	set_variable = { var = pwc_seat_civil_admin_cand value = 1 }
	pwc_se_rebuild_seat_civil_admin = yes
	if = {
		limit = { has_dynamic_modifier = pwc_dm_seat_civil_admin }
		remove_dynamic_modifier = { modifier = pwc_dm_seat_civil_admin }
	}
	add_dynamic_modifier = { modifier = pwc_dm_seat_civil_admin }
	force_update_dynamic_modifier = yes
	add_to_variable = { pwc_seat_civil_admin_ver = 1 }
	set_country_flag = pwc_flag_seat_civil_admin_active
	log = "PWC CABINET appoint seat=seat_civil_admin cand=1 ver=[?ROOT.pwc_seat_civil_admin_ver]"
}
