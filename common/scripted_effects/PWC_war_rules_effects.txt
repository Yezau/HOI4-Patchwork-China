	if = {
		limit = {
			pwc_st_legal_war_request = yes
			NOT = { pwc_st_warning_in_progress = yes }
		}
		set_country_flag = pwc_flag_warning_to_@FROM
		
		set_country_flag = pwc_flag_warning_active
		FROM = { set_country_flag = pwc_flag_warning_from_@ROOT }
		pwc_se_warning_break_nap = yes
		pwc_se_warning_cancel_access = yes
		if = {
			limit = { has_game_rule = { rule = pwc_war_warning option = OFF } }
			pwc_se_on_warning_expiry = yes
		}
		else = {
			pwc_se_add_war_warning_alert = yes
			log = "PWC WARNING start initiator=[ROOT.GetName] target=[FROM.GetName]"
		}
	}
}



pwc_se_add_war_warning_alert = {
	FROM = {
		if = {
			limit = {
				NOT = { has_war_with = ROOT }
				NOT = {
					is_in_array = {
						array = pwc_war_warning_attackers
						value = ROOT
					}
				}
			}
			add_to_array = {
				array = pwc_war_warning_attackers
				value = ROOT
			}
			add_to_variable = { global.pwc_alerts_dirty = 1 }
		}
	}
}


pwc_se_clear_war_warning_alert = {
	if = {
		limit = { FROM = { exists = yes } }
		FROM = {
			if = {
				limit = {
					is_in_array = {
						array = pwc_war_warning_attackers
						value = ROOT
					}
				}
				remove_from_array = {
					array = pwc_war_warning_attackers
					value = ROOT
				}
				add_to_variable = { global.pwc_alerts_dirty = 1 }
			}
		}
	}
}


pwc_se_clear_war_warning = {
	clr_country_flag = pwc_flag_warning_to_@FROM
	if = {
		limit = { FROM = { exists = yes } }
		FROM = { clr_country_flag = pwc_flag_warning_from_@ROOT }
	}
	pwc_se_clear_war_warning_alert = yes
	
	if = {
		limit = {
			NOT = {
				any_country = { has_country_flag = pwc_flag_warning_from_@ROOT }
			}
		}
		clr_country_flag = pwc_flag_warning_active
	}
}


pwc_se_refresh_warning_active = {
    if = {
        limit = {
            has_country_flag = pwc_flag_warning_active
            NOT = {
                any_country = {
                    pwc_st_city_alive = yes
                    has_country_flag = pwc_flag_warning_from_@PREV
                }
            }
        }
        clr_country_flag = pwc_flag_warning_active
    }
}



pwc_se_clear_eliminated_war_warnings = {
    every_possible_country = {
        limit = { pwc_st_is_city_country = yes }
        clr_country_flag = pwc_flag_warning_from_@PREV
        PREV = {
            clr_country_flag = pwc_flag_warning_to_@PREV
            clr_country_flag = pwc_flag_warning_from_@PREV
        }
        if = {
            limit = { has_country_flag = pwc_flag_warning_to_@PREV }
            clr_country_flag = pwc_flag_warning_to_@PREV
            pwc_se_refresh_warning_active = yes
        }
        if = {
            limit = {
                is_in_array = { array = pwc_war_warning_attackers value = PREV }
            }
            remove_from_array = { array = pwc_war_warning_attackers value = PREV }
            add_to_variable = { global.pwc_alerts_dirty = 1 }
        }
    }
    clr_country_flag = pwc_flag_warning_active
    if = {
        limit = { check_variable = { pwc_war_warning_attackers^num > 0 } }
        clear_array = pwc_war_warning_attackers
        add_to_variable = { global.pwc_alerts_dirty = 1 }
    }
}



pwc_se_reconcile_war_warnings = {
    if = {
        limit = {
            OR = {
                has_country_flag = pwc_flag_warning_active
                check_variable = { pwc_war_warning_attackers^num > 0 }
            }
        }
        every_possible_country = {
            limit = {
                pwc_st_is_city_country = yes
                OR = { exists = no pwc_st_city_alive = no }
                OR = {
                    has_country_flag = pwc_flag_warning_from_@PREV
                    has_country_flag = pwc_flag_warning_to_@PREV
                    is_in_array = { array = pwc_war_warning_attackers value = PREV }
                    PREV = {
                        OR = {
                            has_country_flag = pwc_flag_warning_from_@PREV
                            has_country_flag = pwc_flag_warning_to_@PREV
                            is_in_array = {
                                array = pwc_war_warning_attackers
                                value = PREV
                            }
                        }
                    }
                }
            }
            pwc_se_clear_eliminated_war_warnings = yes
        }
        pwc_se_refresh_warning_active = yes
    }
}




pwc_se_clear_integration_alert = {
	clear_variable = integration_alert_timer_@var:target
	clear_variable = integration_progressbar_value_@var:target
	if = {
		limit = {
			is_in_array = {
				array = pwc_state_integration_alerts
				value = var:target
			}
		}
		remove_from_array = {
			array = pwc_state_integration_alerts
			value = var:target
		}
	}
	add_to_variable = { global.pwc_alerts_dirty = 1 }
}

pwc_se_update_integration_progressbar = {
	set_variable = {
		integration_progressbar_value_@var:target = {
			value = integration_alert_timer_@var:target
			divide = 72
			multiply = 300
			subtract = 300
			round = yes
			clamp = { min = -300 max = 0 }
		}
	}
}




pwc_se_push_integration_alert = {
	if = {
		limit = {
			is_ai = no
			NOT = {
				is_in_array = {
					array = pwc_state_integration_alerts
					value = event_target:pwc_cored_state
				}
			}
		}
		set_temp_variable = { target = event_target:pwc_cored_state }
		set_variable = {
			integration_progressbar_value_@var:target = -300
		}
		set_variable = {
			integration_alert_timer_@var:target = 0
		}
		add_to_array = {
			array = pwc_state_integration_alerts
			value = var:target
		}
		add_to_variable = { global.pwc_alerts_dirty = 1 }
		pwc_se_ensure_integration_tick = yes
	}
}



pwc_se_ensure_integration_tick = {
    if = {
        limit = {
            is_ai = no
            check_variable = { pwc_state_integration_alerts^num > 0 }
            NOT = { has_country_flag = pwc_flag_integration_tick_pending }
        }
        set_country_flag = {
            flag = pwc_flag_integration_tick_pending days = 1
        }
        country_event = { id = pwc_ev_integration_tick.2 hours = 1 }
    }
}

pwc_se_tick_integration_alerts = {
    clr_country_flag = pwc_flag_integration_tick_pending
    if = {
        limit = {
            is_ai = no
            check_variable = { pwc_state_integration_alerts^num > 0 }
        }
        
        clear_temp_array = pwc_integration_tick_snapshot
        for_each_loop = {
            array = pwc_state_integration_alerts
            value = pwc_integration_tick_state
            add_to_temp_array = {
                array = pwc_integration_tick_snapshot
                value = var:pwc_integration_tick_state
            }
        }
        for_each_loop = {
            array = pwc_integration_tick_snapshot
            value = target
            add_to_variable = { integration_alert_timer_@var:target = 1 }
            if = {
                limit = {
                    check_variable = { integration_alert_timer_@var:target > 71 }
                }
                pwc_se_clear_integration_alert = yes
            }
            else = { pwc_se_update_integration_progressbar = yes }
        }
        clear_temp_array = pwc_integration_tick_snapshot
        add_to_variable = { global.pwc_alerts_dirty = 1 }
        pwc_se_ensure_integration_tick = yes
    }
}

pwc_se_on_warning_expiry = {
	if = {
		limit = { has_country_flag = pwc_flag_warning_to_@FROM }
		pwc_se_clear_war_warning = yes
		if = {
			limit = { FROM = { exists = yes } }
			remove_targeted_decision = { target = FROM decision = pwc_dec_war_warning_countdown }
		}
		pwc_se_declare_war_on_warning_expiry = yes
		if = {
			limit = {
				FROM = { exists = yes }
				has_war_with = FROM
			}
			log = "PWC WAR declared on warning expiry initiator=[ROOT.GetName] target=[FROM.GetName]"
		}
		else = {
			log = "PWC WARNING expired without war initiator=[ROOT.GetName]"
		}
	}
}




pwc_se_on_surrender_settle = {
	if = {
		limit = {
			pwc_st_is_city_country = yes
			pwc_st_settlement_not_done = yes
			NOT = { has_game_rule = { rule = pwc_peace_conference option = ON } }
		}
		set_country_flag = pwc_flag_settled
		pwc_se_capture_war_report = yes
		add_to_variable = { global.pwc_capitulation_fires = 1 }
		add_to_variable = { global.pwc_settle_runs = 1 }
		log = "PWC SETTLE begin target=[ROOT.GetName] fires=[?global.pwc_capitulation_fires]"

		
		set_variable = { var = pwc_best_gain value = -1 }
		set_variable = { var = pwc_best_rank value = 999 }
		save_event_target_as = pwc_capital_controller
		save_event_target_as = pwc_claimant

		
		
		
		
		every_enemy_country = {
			set_variable = { var = pwc_war_gain value = 0 }
			every_state = {
				limit = {
					is_owned_by = ROOT
					is_controlled_by = PREV
				}
				transfer_state_to = PREV
				PREV = { add_to_variable = { pwc_war_gain = 1 } }
				if = {
					limit = { is_capital = yes }
					PREV = { save_event_target_as = pwc_capital_controller }
				}
			}
		}

		
		
		every_enemy_country = {
			if = {
				limit = { check_variable = { var = pwc_war_gain value = ROOT.pwc_best_gain compare = greater_than } }
				set_variable = { var = ROOT.pwc_best_gain value = pwc_war_gain }
			}
		}
		every_enemy_country = {
			if = {
				limit = {
					pwc_st_city_alive = yes
					check_variable = { var = pwc_war_gain value = ROOT.pwc_best_gain compare = equals }
					check_variable = { var = pwc_tag_rank value = ROOT.pwc_best_rank compare = less_than }
				}
				set_variable = { var = ROOT.pwc_best_rank value = pwc_tag_rank }
				save_event_target_as = pwc_claimant
			}
		}

		
		
		
		
		
		
		
		
		every_state = {
			limit = { is_owned_by = ROOT }
			if = {
				limit = { NOT = { event_target:pwc_capital_controller = { tag = ROOT } } }
				transfer_state_to = event_target:pwc_capital_controller
				log = "PWC SETTLE residual -> capital controller"
			}
			else_if = {
				limit = { NOT = { event_target:pwc_claimant = { tag = ROOT } } }
				transfer_state_to = event_target:pwc_claimant
				log = "PWC SETTLE residual -> claimant"
			}
		}

		
		
		every_enemy_country = { pwc_se_apply_war_cooldown = yes }
		
		every_enemy_country = { white_peace = { tag = ROOT } }

		
		pwc_se_record_settlement_report_winner = yes
		pwc_se_mark_city_eliminated = yes
		if = {
			limit = { NOT = { event_target:pwc_capital_controller = { tag = ROOT } } }
			event_target:pwc_capital_controller = { annex_country = { target = ROOT transfer_troops = yes } }
			log = "PWC SETTLE remnant annexed by capital controller"
		}
		else_if = {
			limit = { NOT = { event_target:pwc_claimant = { tag = ROOT } } }
			event_target:pwc_claimant = { annex_country = { target = ROOT transfer_troops = yes } }
			log = "PWC SETTLE remnant annexed by claimant"
		}
		log = "PWC SETTLE end runs=[?global.pwc_settle_runs]"
	}
	else = {
		log = "PWC SETTLE skipped, already settled"
	}
}
