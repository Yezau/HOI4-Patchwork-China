pwc_se_cataclysm_initialize_offsite = {
    if = {
        limit = { NOT = { has_country_flag = pwc_cataclysm_offsite_initialized } }
        set_country_flag = pwc_cataclysm_offsite_initialized
        set_variable = { pwc_cataclysm_offsite_civ = 0 }
        set_variable = { pwc_cataclysm_offsite_mil = 0 }
        set_variable = { pwc_cataclysm_offsite_dock = 0 }
    }
}
pwc_se_cataclysm_change_offsite_civ = {
    pwc_se_cataclysm_initialize_offsite = yes
    set_variable = { pwc_cataclysm_offsite_target = pwc_cataclysm_offsite_civ }
    add_to_variable = { pwc_cataclysm_offsite_target = pwc_cataclysm_offsite_delta }
    clamp_variable = { var = pwc_cataclysm_offsite_target min = 0 }
    set_variable = { pwc_cataclysm_offsite_applied = pwc_cataclysm_offsite_target }
    subtract_from_variable = { pwc_cataclysm_offsite_applied = pwc_cataclysm_offsite_civ }
    add_offsite_building = { type = industrial_complex level = var:pwc_cataclysm_offsite_applied }
    set_variable = { pwc_cataclysm_offsite_civ = pwc_cataclysm_offsite_target }
    clear_variable = pwc_cataclysm_offsite_target
    clear_variable = pwc_cataclysm_offsite_applied
}

pwc_se_cataclysm_refresh_membership = {
    clear_array = global.pwc_cataclysm_humans
    clear_array = global.pwc_cataclysm_protected_humans
    clear_array = global.pwc_cataclysm_pools
    for_each_scope_loop = {
        array = global.pwc_cataclysm_cities
        clear_array = pwc_cataclysm_members
        
        if = {
            limit = { exists = yes is_ai = no }
            add_to_array = { array = global.pwc_cataclysm_protected_humans value = THIS }
        }
        if = {
            limit = { pwc_st_cataclysm_human = yes }
            add_to_array = { array = global.pwc_cataclysm_humans value = THIS }
        }
        else_if = {
            limit = { has_variable = pwc_cataclysm_pool }
            var:pwc_cataclysm_pool = { pwc_se_cataclysm_invalidate_pool = yes }
            clear_variable = pwc_cataclysm_pool
        }
    }
    
    for_each_scope_loop = {
        array = global.pwc_cataclysm_humans
        save_event_target_as = pwc_cataclysm_human
        set_variable = { pwc_cataclysm_proposed_pool = THIS }
        clr_country_flag = pwc_cataclysm_pool_selected
        for_each_scope_loop = {
            array = global.pwc_cataclysm_humans
            if = {
                limit = {
                    is_in_faction_with = event_target:pwc_cataclysm_human
                    event_target:pwc_cataclysm_human = {
                        NOT = { has_country_flag = pwc_cataclysm_pool_selected }
                    }
                    check_variable = {
                        var = pwc_cataclysm_rank
                        value = PREV.pwc_cataclysm_rank
                        compare = less_than
                    }
                }
                event_target:pwc_cataclysm_human = {
                    set_variable = { pwc_cataclysm_proposed_pool = PREV }
                    set_country_flag = pwc_cataclysm_pool_selected
                }
            }
        }
        if = {
            limit = {
                OR = {
                    NOT = { has_variable = pwc_cataclysm_pool }
                    NOT = {
                        check_variable = {
                            pwc_cataclysm_pool = pwc_cataclysm_proposed_pool
                        }
                    }
                }
            }
            if = {
                limit = { has_variable = pwc_cataclysm_pool }
                var:pwc_cataclysm_pool = { pwc_se_cataclysm_invalidate_pool = yes }
            }
            set_variable = { pwc_cataclysm_pool = pwc_cataclysm_proposed_pool }
            var:pwc_cataclysm_pool = { pwc_se_cataclysm_invalidate_pool = yes }
        }
        var:pwc_cataclysm_pool = {
            add_to_array = { array = pwc_cataclysm_members value = PREV }
            if = {
                limit = { PREV = { has_country_flag = pwc_cataclysm_arrival_sent } }
                set_country_flag = pwc_cataclysm_arrival_sent
            }
        }
        if = {
            limit = { check_variable = { pwc_cataclysm_pool = THIS } }
            add_to_array = { array = global.pwc_cataclysm_pools value = THIS }
        }
    }
    
    
    for_each_scope_loop = {
        array = global.pwc_cataclysm_humans
        if = {
            limit = {
                var:pwc_cataclysm_pool = {
                    has_country_flag = pwc_cataclysm_arrival_sent
                }
            }
            set_country_flag = pwc_cataclysm_arrival_sent
        }
    }
    pwc_se_cataclysm_resolve_budget_pools = yes
}

                                pwc_cataclysm_budget_pool = PREV.pwc_cataclysm_pool
                            }

                                NOT = {
                                    is_in_array = {
                                        array = pwc_cataclysm_continuing_pools
                                        value = PREV.pwc_cataclysm_pool
                                    }
                                }
                            }

                                array = pwc_cataclysm_continuing_pools
                                value = PREV.pwc_cataclysm_pool
                            }

        }
    }

            add_to_variable = { pwc_cataclysm_m0 = PREV.pwc_cataclysm_raw_mil }
        }
    }
    add_to_variable = { pwc_cataclysm_samples = 1 }
    clamp_variable = { var = pwc_cataclysm_samples min = 0 max = 3 }
    set_variable = { pwc_cataclysm_average = pwc_cataclysm_f0 }

    add_to_variable = { pwc_cataclysm_average = pwc_cataclysm_f2 }
    divide_variable = { pwc_cataclysm_average = 3 }
    if = {
        limit = {
            OR = {
                check_variable = { pwc_cataclysm_samples < 3 }
                check_variable = { pwc_cataclysm_f0 < 40 }
                check_variable = { pwc_cataclysm_m0 < 8 }
            }

    }
    else = { clr_country_flag = pwc_cataclysm_paused }

        limit = {
            check_variable = { pwc_cataclysm_samples > 2 }

                var = pwc_cataclysm_average
                value = pwc_cataclysm_decline_threshold
                compare = less_than
            }
        }

    }

        set_variable = { pwc_cataclysm_decline_samples = 0 }
        set_variable = { pwc_cataclysm_stage_day = global.num_days }
        pwc_se_cataclysm_set_limits = yes
    }

                clr_country_flag = pwc_cataclysm_transport_ready
                if = {
                    limit = { pwc_st_cataclysm_transport_ready = yes }
                    set_country_flag = pwc_cataclysm_transport_ready
                }
                for_each_scope_loop = {
                    array = owned_states
                    if = {
                        limit = {
                            event_target:pwc_cataclysm_recipient = {
                                NOT = { has_country_flag = pwc_cataclysm_found_rear }
                                has_country_flag = pwc_cataclysm_transport_ready
                            }
                        }
                        if = {
                            limit = { pwc_st_cataclysm_rear = yes }
                            save_event_target_as = pwc_cataclysm_spawn_state
                            event_target:pwc_cataclysm_recipient = {
                                set_country_flag = pwc_cataclysm_found_rear
                            }
                        }
                    }
                }
                if = {
                    limit = { has_country_flag = pwc_cataclysm_found_rear }
                    set_variable = {
                        pwc_cataclysm_next_heavy =
                        PREV.pwc_cataclysm_composition
                    }
                    modulo_variable = { pwc_cataclysm_next_heavy = 4 }
                    if = {
                        limit = { check_variable = { pwc_cataclysm_next_heavy = 3 } }
                        set_variable = { pwc_cataclysm_next_heavy = 1 }
                    }
                    else = { set_variable = { pwc_cataclysm_next_heavy = 0 } }
                    pwc_se_cataclysm_spawn_one = yes
                    if = {
                        limit = { has_country_flag = pwc_cataclysm_spawn_success }
                        add_to_variable = { global.pwc_cataclysm_used = 1 }
                        add_to_variable = { global.pwc_cataclysm_live = 1 }
                        var:pwc_cataclysm_owner = {
                            add_to_variable = { pwc_cataclysm_source_used = 1 }
                            add_to_variable = { pwc_cataclysm_source_weekly_used = 1 }
                        }
                        
                        if = {
                            limit = {
                                var:pwc_cataclysm_owner = {
                                    NOT = { tag = event_target:pwc_cataclysm_pool }
                                }
                            }
                            var:pwc_cataclysm_owner = {
                                add_to_variable = { pwc_cataclysm_used = 1 }
                                add_to_variable = { pwc_cataclysm_live_pool = 1 }
                            }
                        }
                        event_target:pwc_cataclysm_pool = {
                            add_to_variable = { pwc_cataclysm_used = 1 }
                            add_to_variable = { pwc_cataclysm_live_pool = 1 }
                            add_to_variable = { pwc_cataclysm_composition = 1 }
                            add_to_variable = { pwc_cataclysm_dispatch_success = 1 }
                        }
                    }
                }
            }
        }
    }
}
pwc_se_cataclysm_dispatch_pool = {
    set_variable = { pwc_cataclysm_dispatch_success = 0 }
    if = {
        limit = {
            NOT = { has_country_flag = pwc_cataclysm_arrival_sent }
            check_variable = { pwc_cataclysm_candidates^num > 0 }
            pwc_st_cataclysm_budget_available = yes
        }
        
        set_country_flag = pwc_cataclysm_arrival_sent
        for_each_scope_loop = {
            array = pwc_cataclysm_members
            set_country_flag = pwc_cataclysm_arrival_sent
        }
        for_loop_effect = {
            start = 0 end = pwc_cataclysm_arrival_budget
            pwc_se_cataclysm_attempt_one = yes
        }
        add_to_variable = { pwc_cataclysm_weekly_used = pwc_cataclysm_dispatch_success }
    }
    else_if = {
        limit = {
            check_variable = {
                var = pwc_cataclysm_weekly_used
                value = pwc_cataclysm_weekly_budget
                compare = less_than
            }
            check_variable = { global.pwc_cataclysm_week_remainder = 0 }
        }
        set_variable = { pwc_cataclysm_dispatch_allowance = pwc_cataclysm_weekly_budget }
        subtract_from_variable = {
            pwc_cataclysm_dispatch_allowance = pwc_cataclysm_weekly_used
        }
        for_loop_effect = {
            start = 0 end = pwc_cataclysm_dispatch_allowance
            pwc_se_cataclysm_attempt_one = yes
        }
        add_to_variable = { pwc_cataclysm_weekly_used = pwc_cataclysm_dispatch_success }
    }
}
pwc_se_cataclysm_global_daily = {
    if = {
        limit = {
            pwc_st_cataclysm_enabled = yes
            has_global_flag = pwc_cataclysm_initialized
            NOT = {
                check_variable = { global.pwc_cataclysm_last_day = global.num_days }
            }
        }
        set_variable = { global.pwc_cataclysm_last_day = global.num_days }
        pwc_se_cataclysm_clock = yes
        pwc_se_cataclysm_reset_cycle = yes
        pwc_se_cataclysm_reset_week = yes
        set_variable = { global.pwc_cataclysm_week_remainder = global.pwc_cataclysm_day }
        modulo_variable = { global.pwc_cataclysm_week_remainder = 7 }
        pwc_se_cataclysm_refresh_membership = yes
        
        set_global_flag = pwc_cataclysm_relation_batch
        pwc_se_cataclysm_cleanup_holders = yes
        for_each_scope_loop = {
            array = global.pwc_cataclysm_recipients
            clr_country_flag = pwc_cataclysm_qualified
            clr_country_flag = pwc_cataclysm_candidate
            clear_array = pwc_cataclysm_front_pools
        }
        for_each_scope_loop = {
            array = global.pwc_cataclysm_pools
            clear_array = pwc_cataclysm_candidates
            pwc_se_cataclysm_progress_pool = yes
        }
        
        for_each_scope_loop = {
            array = global.pwc_cataclysm_pools
            if = {
                limit = { has_country_flag = pwc_cataclysm_active }
                pwc_se_cataclysm_collect_candidates = yes
            }
            else = { clear_array = pwc_cataclysm_candidates }
        }
        for_each_scope_loop = {
            array = global.pwc_cataclysm_recipients
            if = {
                limit = { has_variable = pwc_cataclysm_owner }
                pwc_se_cataclysm_recipient_cleanup = yes
            }
        }
        clr_global_flag = pwc_cataclysm_dispatch_due
        for_each_scope_loop = {
            array = global.pwc_cataclysm_pools
            if = {
                limit = {
                    pwc_st_cataclysm_pool_active = yes
                    check_variable = { pwc_cataclysm_candidates^num > 0 }
                    OR = {
                        NOT = { has_country_flag = pwc_cataclysm_arrival_sent }
                        check_variable = { global.pwc_cataclysm_week_remainder = 0 }
                    }
                }
                set_global_flag = pwc_cataclysm_dispatch_due
            }
        }
        if = {
            limit = { has_global_flag = pwc_cataclysm_dispatch_due }
            
            pwc_se_cataclysm_recount_all = yes
            for_each_scope_loop = {
                array = global.pwc_cataclysm_recipients
                if = {
                    limit = { has_country_flag = pwc_cataclysm_qualified }
                    pwc_se_cataclysm_set_recipient_cap = yes
                }
            }
            for_each_scope_loop = {
                array = global.pwc_cataclysm_pools
                if = {
                    limit = { pwc_st_cataclysm_pool_active = yes }
                    pwc_se_cataclysm_dispatch_pool = yes
                }
            }
        }
        for_each_scope_loop = {
            array = global.pwc_cataclysm_relation_touched
            clr_country_flag = pwc_cataclysm_relation_checked
        }
        clear_array = global.pwc_cataclysm_relation_touched
        clr_global_flag = pwc_cataclysm_relation_batch
    }
}
