pwc_se_cataclysm_recount_units = {
    set_variable = { pwc_cataclysm_live = 0 }
    every_country_division = {
        limit = {
            OR = {
                division_has_majority_template = pwc_unit_cataclysm_assault
                division_has_majority_template = pwc_unit_cataclysm_heavy
            }
        }
        PREV = {
            add_to_variable = { pwc_cataclysm_live = 1 }
        }
    }
    set_variable = { pwc_cataclysm_regular_divisions = num_divisions }
    subtract_from_variable = { pwc_cataclysm_regular_divisions = pwc_cataclysm_live }
    clamp_variable = { var = pwc_cataclysm_regular_divisions min = 0 }
    if = {
        limit = { check_variable = { var = pwc_cataclysm_live value = 1 compare = greater_than_or_equals } }
        set_country_flag = pwc_cataclysm_has_units
    }
    else = {
        clr_country_flag = pwc_cataclysm_has_units
    }
}
