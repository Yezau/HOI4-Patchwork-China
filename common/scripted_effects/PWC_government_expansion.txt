pwc_se_init_institutions = {
    if = {
        limit = { NOT = { has_country_flag = pwc_institutions_initialized } }
        if = {
            limit = { NOT = { OR = { has_idea = pwc_institution_administration_local has_idea = pwc_institution_administration_coordinated has_idea = pwc_institution_administration_central } } }
            add_ideas = pwc_institution_administration_coordinated
        }
        if = {
            limit = { NOT = { OR = { has_idea = pwc_institution_labor_protected has_idea = pwc_institution_labor_negotiated has_idea = pwc_institution_labor_war } } }
            add_ideas = pwc_institution_labor_negotiated
        }
        if = {
            limit = { NOT = { OR = { has_idea = pwc_institution_education_basic has_idea = pwc_institution_education_vocational has_idea = pwc_institution_education_military } } }
            add_ideas = pwc_institution_education_vocational
        }
        set_country_flag = pwc_institutions_initialized
    }
}
pwc_se_start_pwc_agenda_civic_relief_50 = {
    if = {
        limit = { pwc_st_agenda_civic_relief_eligible = yes NOT = { pwc_st_agenda_civic_relief_discount = yes } }
        set_variable = { pwc_agenda_paid = 50 }
        if = {
            limit = { pwc_st_agenda_civic_relief_audit = yes }
            add_timed_idea = { idea = pwc_agenda_civic_relief_audited days = 90 }
        }
        else = {
            add_timed_idea = { idea = pwc_agenda_civic_relief_normal days = 90 }
        }
        if = {
            limit = { is_ai = no }
            country_event = { id = pwc_government.1 }
        }
    }
}
