pwc_se_officer_grant_admiral_t1 = {
    if = {
        limit = { NOT = { has_country_flag = pwc_flag_officer_paid_admiral_t1 } }
        generate_character = { token_base = pwc_admiral_t1 }
        every_character = {
            limit = { is_character = pwc_admiral_t1 }
            if = {
                limit = { is_unit_leader = no }
                pwc_se_officer_assign_admiral = yes
            }
            if = {
                limit = { is_unit_leader = yes }
                PREV = { set_country_flag = pwc_flag_officer_paid_admiral_t1 }
            }
        }
    }
}
