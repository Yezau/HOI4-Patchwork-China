









pwc_se_rebuild_law_conscription = {
	if = {
		limit = { check_variable = { var = pwc_law_conscription value = 1 compare = equals } }
		set_variable = { var = pwc_law_conscription_v_conscription value = 0.01 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = 0.05 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0 }
	}
	else_if = {
		limit = { check_variable = { var = pwc_law_conscription value = 2 compare = equals } }
		set_variable = { var = pwc_law_conscription_v_conscription value = 0.015 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = 0.03 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0 }
	}
	else_if = {
		limit = { check_variable = { var = pwc_law_conscription value = 3 compare = equals } }
		set_variable = { var = pwc_law_conscription_v_conscription value = 0.025 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = 0 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0 }
	}
	else_if = {
		limit = { check_variable = { var = pwc_law_conscription value = 4 compare = equals } }
		set_variable = { var = pwc_law_conscription_v_conscription value = 0.05 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = -0.05 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0.05 }
	}
	else_if = {
		limit = { check_variable = { var = pwc_law_conscription value = 5 compare = equals } }
		set_variable = { var = pwc_law_conscription_v_conscription value = 0.1 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = -0.1 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0.1 }
	}
	else = {
		set_variable = { var = pwc_law_conscription_v_conscription value = 0 }
		set_variable = { var = pwc_law_conscription_v_stability_factor value = 0 }
		set_variable = { var = pwc_law_conscription_v_training_time_factor value = 0 }
	}
}


pwc_se_sync_law_conscription_1 = {
	if = {
		limit = {
			NOT = { check_variable = { var = pwc_law_conscription value = 1 compare = equals } }
		}
		add_to_variable = { pwc_law_conscription_ver = 1 }
	}
	set_variable = { var = pwc_law_conscription value = 1 }
	pwc_se_rebuild_law_conscription = yes
	log = "PWC LAW set law=conscription level=1 ver=[?ROOT.pwc_law_conscription_ver]"
}
