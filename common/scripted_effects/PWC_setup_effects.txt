pwc_se_init_setup_country = {
    if = {
        limit = {
            pwc_st_is_city_country = yes
            NOT = { has_country_flag = pwc_flag_setup_country_applied }
        }
        set_country_flag = pwc_flag_setup_country_applied
        pwc_se_assign_setup_archetypes = yes
        pwc_se_calibrate_starting_industry = yes
        pwc_se_apply_integration_speed = yes
        pwc_se_refresh_setup_country = yes
    }
}
pwc_se_refresh_setup_country = {
    if = {
        limit = {
            pwc_st_is_city_country = yes
            pwc_st_city_alive = yes
            exists = yes
        }
        pwc_se_update_administration = yes
        pwc_se_update_unit_limit = yes
        pwc_se_update_demobilization = yes
        pwc_se_monthly_gate_latch = yes
        pwc_se_arm_war_phase = yes
        pwc_se_refresh_regional_identity = yes
    }
}
pwc_se_update_administration = {
    set_variable = { pwc_administrative_pressure = 0 }
    if = {
        limit = {
            pwc_st_is_city_country = yes
            NOT = {
                has_game_rule = { rule = pwc_administrative_pressure option = REMOVE }
            }
        }
        every_owned_state = {
            limit = { is_capital = no }
            if = {
                limit = { is_core_of = PREV }
                PREV = {
                    add_to_variable = { pwc_administrative_pressure = 0.002 }
                }
            }
            else = {
                PREV = {
                    add_to_variable = { pwc_administrative_pressure = 0.005 }
                }
            }
        }
        clamp_variable = {
            var = pwc_administrative_pressure min = 0 max = 0.1
        }
        if = {
            limit = {
                NOT = {
                    has_dynamic_modifier = { modifier = pwc_dm_administrative_pressure }
                }
            }
            add_dynamic_modifier = { modifier = pwc_dm_administrative_pressure }
        }
    }
    else = {
        remove_dynamic_modifier = { modifier = pwc_dm_administrative_pressure }
    }
}
pwc_se_update_unit_limit = {
    set_variable = { pwc_setup_owned_states = num_owned_states }
    set_variable = { pwc_setup_controlled_states = 0 }
    every_owned_state = {
        limit = { is_fully_controlled_by = PREV }
        PREV = { add_to_variable = { pwc_setup_controlled_states = 1 } }
    }
    set_variable = {
        pwc_unit_capacity = {
            value = pwc_setup_controlled_states
            multiply = 4
            add = 8
            clamp = { min = 8 max = 500 }
        }
    }
    set_variable = { pwc_unit_penalty = 0 }
    set_variable = { pwc_unit_penalty_negative = 0 }
    set_variable = { pwc_unit_ai_limit = 0 }
    set_variable = { pwc_unit_capacity_bar_offset = 0 }
    set_variable = { pwc_unit_penalty_percent = 0 }
    if = {
        limit = { pwc_st_unit_limit_applies = yes }
        set_variable = {
            pwc_unit_penalty = {
                value = num_divisions
                divide = pwc_unit_capacity
                subtract = 1
                multiply = 0.25
                clamp = { min = 0 max = 0.25 }
            }
        }
        set_variable = {
            pwc_unit_penalty_negative = { value = pwc_unit_penalty multiply = -1 }
        }
        set_variable = {
            pwc_unit_capacity_bar_offset = {
                value = num_divisions
                divide = pwc_unit_capacity
                multiply = 300
                subtract = 300
                round = yes
                clamp = { min = -300 max = 0 }
            }
        }
        set_variable = {
            pwc_unit_penalty_percent = {
                value = pwc_unit_penalty
                multiply = 100
                round = yes
            }
        }
        if = {
            limit = {
                check_variable = {
                    var = num_divisions
                    value = pwc_unit_capacity
                    compare = greater_than_or_equals
                }
            }
            set_variable = { pwc_unit_ai_limit = -1 }
        }
        if = {
            limit = {
                NOT = { has_dynamic_modifier = { modifier = pwc_dm_unit_capacity } }
            }
            add_dynamic_modifier = { modifier = pwc_dm_unit_capacity }
        }
    }
    else = { remove_dynamic_modifier = { modifier = pwc_dm_unit_capacity } }
}
pwc_se_update_demobilization = {
    if = {
        limit = {
            pwc_st_demobilization_applies = yes
            has_war = no
            OR = {
                check_variable = { pwc_law_conscription > 3 }
                check_variable = {
                    pwc_law_economy_mobilization > 3
                }
            }
            NOT = { has_active_mission = pwc_demobilization_grace }
        }
        activate_mission = pwc_demobilization_grace
    }
}
pwc_se_demobilize = {
    if = {
        limit = { pwc_st_demobilization_applies = yes has_war = no }
        if = {
            limit = { check_variable = { pwc_law_conscription > 3 } }
            pwc_se_set_law_conscription_3 = yes
        }
        if = {
            limit = {
                check_variable = {
                    pwc_law_economy_mobilization > 3
                }
            }
            pwc_se_set_law_economy_mobilization_3 = yes
        }
    }
}
pwc_se_calibrate_starting_industry = {
    if = {
        limit = {
            pwc_st_is_city_country = yes
            NOT = { has_country_flag = pwc_flag_setup_industry_applied }
            OR = {
                has_game_rule = { rule = pwc_industry_calibration option = NORMALIZE }
                has_game_rule = { rule = pwc_industry_calibration option = BALANCE }
            }
        }
        set_country_flag = pwc_flag_setup_industry_applied
        
        
        set_variable = { pwc_setup_offsite_civ = num_of_civilian_factories }
        set_variable = { pwc_setup_offsite_mil = num_of_military_factories }
        every_owned_state = {
            PREV = {
                subtract_from_variable = {
                    pwc_setup_offsite_civ = PREV.industrial_complex_level
                }
                subtract_from_variable = {
                    pwc_setup_offsite_mil = PREV.arms_factory_level
                }
            }
        }
        clamp_variable = { var = pwc_setup_offsite_civ min = 0 }
        clamp_variable = { var = pwc_setup_offsite_mil min = 0 }
        multiply_variable = { pwc_setup_offsite_civ = -1 }
        multiply_variable = { pwc_setup_offsite_mil = -1 }
        add_offsite_building = {
            type = industrial_complex level = var:pwc_setup_offsite_civ
        }
        add_offsite_building = { type = arms_factory level = var:pwc_setup_offsite_mil }
        clear_variable = pwc_setup_offsite_civ
        clear_variable = pwc_setup_offsite_mil
        if = {
            limit = {
                has_game_rule = { rule = pwc_industry_calibration option = BALANCE }
            }
            every_owned_state = {
                set_building_level = { type = industrial_complex level = 2 }
                set_building_level = { type = arms_factory level = 1 }
            }
        }
    }
}

pwc_se_apply_integration_speed = {
    THIS = {
        if = {
            limit = { has_game_rule = { rule = pwc_integration_speed option = FAST } }
            set_occupation_law = pwc_city_administration_fast
        }
        else_if = {
            limit = { has_game_rule = { rule = pwc_integration_speed option = STRICT } }
            set_occupation_law = pwc_city_administration_strict
        }
        else = { set_occupation_law = pwc_city_military_administration }
    }
}

            if = {
                limit = {
                    pwc_st_is_city_country = yes

                        has_game_rule = { rule = pwc_integration_speed option = FAST }
                        has_game_rule = { rule = pwc_integration_speed option = STRICT }
                    } }
                }
                if = {
                    limit = {
                        OR = {
                            occupation_law = pwc_city_administration_fast
                            occupation_law = pwc_city_administration_strict

                    set_occupation_law = pwc_city_military_administration

                every_occupied_country = {
