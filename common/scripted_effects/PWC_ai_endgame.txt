pwc_se_coalition_track_expansion = {
    if = {
        limit = { is_ai = no pwc_st_city_alive = yes pwc_st_is_city_country = yes }
        if = {
            limit = { NOT = { has_variable = pwc_coalition_states_5 } }
            set_variable = { pwc_coalition_states_0 = num_owned_states }
            set_variable = { pwc_coalition_states_1 = num_owned_states }
            set_variable = { pwc_coalition_states_2 = num_owned_states }
            set_variable = { pwc_coalition_states_3 = num_owned_states }
            set_variable = { pwc_coalition_states_4 = num_owned_states }
            set_variable = { pwc_coalition_states_5 = num_owned_states }
        }
        set_variable = { pwc_coalition_expansion = num_owned_states }
        subtract_from_variable = { pwc_coalition_expansion = pwc_coalition_states_5 }
        set_variable = { pwc_coalition_states_5 = pwc_coalition_states_4 }
        set_variable = { pwc_coalition_states_4 = pwc_coalition_states_3 }
        set_variable = { pwc_coalition_states_3 = pwc_coalition_states_2 }
        set_variable = { pwc_coalition_states_2 = pwc_coalition_states_1 }
        set_variable = { pwc_coalition_states_1 = pwc_coalition_states_0 }
        set_variable = { pwc_coalition_states_0 = num_owned_states }
    }
    else = {
        clear_variable = pwc_coalition_states_0
        clear_variable = pwc_coalition_states_1
        clear_variable = pwc_coalition_states_2
        clear_variable = pwc_coalition_states_3
        clear_variable = pwc_coalition_states_4
        clear_variable = pwc_coalition_states_5
        clear_variable = pwc_coalition_expansion
    }
}
pwc_se_coalition_selection_gaps = {
    set_variable = { pwc_coalition_best_score = -1 }
    set_variable = { pwc_coalition_best_manpower = -1 }
    set_variable = { pwc_coalition_gap_manpower = pwc_coalition_required_manpower }
    subtract_from_variable = { pwc_coalition_gap_manpower = pwc_coalition_manpower }
    clamp_variable = { var = pwc_coalition_gap_manpower min = 0 }
    set_variable = { pwc_coalition_scale_manpower = pwc_coalition_required_manpower }
    clamp_variable = { var = pwc_coalition_scale_manpower min = 1 }
    set_variable = { pwc_coalition_gap_industry = pwc_coalition_required_industry }
    subtract_from_variable = { pwc_coalition_gap_industry = pwc_coalition_industry }
    clamp_variable = { var = pwc_coalition_gap_industry min = 0 }
    set_variable = { pwc_coalition_scale_industry = pwc_coalition_required_industry }
    clamp_variable = { var = pwc_coalition_scale_industry min = 1 }
}
pwc_se_coalition_score_candidate = {
    set_variable = { pwc_coalition_candidate_score = 0 }
    set_variable = { pwc_coalition_contribution = deployed_army_manpower_k }
    clamp_variable = { var = pwc_coalition_contribution min = 0 max = PREV.pwc_coalition_gap_manpower }
    divide_variable = { pwc_coalition_contribution = PREV.pwc_coalition_scale_manpower }
    add_to_variable = { pwc_coalition_candidate_score = pwc_coalition_contribution }
    set_variable = { pwc_coalition_contribution = num_of_factories }
    clamp_variable = { var = pwc_coalition_contribution min = 0 max = PREV.pwc_coalition_gap_industry }
    divide_variable = { pwc_coalition_contribution = PREV.pwc_coalition_scale_industry }
    add_to_variable = { pwc_coalition_candidate_score = pwc_coalition_contribution }
    clear_variable = pwc_coalition_contribution
}
pwc_se_coalition_reset_totals = {
    set_variable = { pwc_coalition_count = 0 }
    set_variable = { pwc_coalition_manpower = 0 }
    set_variable = { pwc_coalition_industry = 0 }
    set_variable = { pwc_coalition_required_manpower = deployed_army_manpower_k }
    multiply_variable = { pwc_coalition_required_manpower = 0.8 }
    set_variable = { pwc_coalition_required_industry = num_of_factories }
    multiply_variable = { pwc_coalition_required_industry = 0.6 }
}


pwc_se_coalition_count_member = {
    event_target:pwc_coalition_player = {
        add_to_variable = { pwc_coalition_count = 1 }
        add_to_variable = { pwc_coalition_manpower = PREV.deployed_army_manpower_k }
        add_to_variable = { pwc_coalition_industry = PREV.num_of_factories }
    }
}

pwc_se_coalition_prepare_member = {
    pwc_se_coalition_reserve = yes
    set_variable = { pwc_coalition_target = event_target:pwc_coalition_player }
}







pwc_se_coalition_release_member = {
    if = {
        limit = { has_variable = pwc_coalition_target }
        
        if = {
            limit = {
                OR = {
                    is_ai = no
                    NOT = { has_war_with = var:pwc_coalition_target }
                }
            }
            clear_variable = pwc_coalition_target
        }
    }
    clr_country_flag = pwc_flag_coalition_reserved
    remove_dynamic_modifier = { modifier = pwc_dm_coalition_muster }
}

pwc_se_coalition_cleanup = {
    save_event_target_as = pwc_coalition_player
    for_each_scope_loop = {
        array = pwc_coalition_members
        if = {
            limit = {
                has_variable = pwc_coalition_target
                var:pwc_coalition_target = { tag = event_target:pwc_coalition_player }
            }
            pwc_se_coalition_release_member = yes
        }
    }
    clear_array = pwc_coalition_members
    clr_country_flag = pwc_flag_coalition_pending
    clr_country_flag = pwc_flag_coalition_warning
    remove_dynamic_modifier = { modifier = pwc_dm_coalition_warning }
}

pwc_se_coalition_weekly = {
    if = {
        limit = { has_country_flag = pwc_flag_coalition_pending }
        save_event_target_as = pwc_coalition_player
        if = {
            limit = { pwc_st_coalition_player = no }
            pwc_se_coalition_cleanup = yes
        }
        else_if = {
            limit = { NOT = { has_country_flag = pwc_flag_coalition_warning } }
            pwc_se_coalition_reset_totals = yes
            clear_array = pwc_coalition_valid_members
            for_each_scope_loop = {
                array = pwc_coalition_members
                if = {
                    limit = {
                        has_variable = pwc_coalition_target
                        var:pwc_coalition_target = {
                            tag = event_target:pwc_coalition_player
                        }
                        pwc_st_coalition_partner = yes
                    }
                    pwc_se_coalition_count_member = yes
                    event_target:pwc_coalition_player = {
                        add_to_array = { array = pwc_coalition_valid_members value = PREV }
                    }
                }
            }
            if = {
                limit = { pwc_st_coalition_size_ready = yes }
                pwc_se_coalition_cooldown = yes
                for_each_scope_loop = {
                    array = pwc_coalition_valid_members
                    if = {
                        limit = {
                            NOT = { has_war_with = event_target:pwc_coalition_player }
                        }
                        declare_war_on = {
                            target = event_target:pwc_coalition_player
                            type = annex_everything
                        }
                    }
                }
            }
            else = {
                pwc_se_coalition_aborted = yes
            }
            clear_array = pwc_coalition_valid_members
            pwc_se_coalition_cleanup = yes
        }
    }
    
    if = {
        limit = {
            has_variable = pwc_coalition_target
            OR = {
                is_ai = no
                AND = {
                    NOT = { has_country_flag = pwc_flag_coalition_reserved }
                    NOT = { has_war_with = var:pwc_coalition_target }
                }
            }
        }
        pwc_se_coalition_release_member = yes
    }
}

pwc_se_ai_finish_war_weekly = {
    if = {
        limit = { pwc_st_ai_only_war = yes }
        if = {
            limit = { NOT = { has_country_flag = pwc_flag_ai_war_tracked } }
            set_country_flag = pwc_flag_ai_war_tracked
            set_country_flag = {
                flag = pwc_flag_ai_war_grace days = 120 value = 1
            }
        }
        if = {
            limit = { pwc_st_ai_finish_war = yes }
            add_dynamic_modifier = { modifier = pwc_dm_ai_sustained_offensive }
        }
        else = {
            remove_dynamic_modifier = { modifier = pwc_dm_ai_sustained_offensive }
        }
    }
    else = {
        clr_country_flag = pwc_flag_ai_war_tracked
        clr_country_flag = pwc_flag_ai_war_grace
        remove_dynamic_modifier = { modifier = pwc_dm_ai_sustained_offensive }
    }
}
