pwc_st_is_city_country = {
    hidden_trigger = { NOT = { tag = NTR } }
}


pwc_st_protection_active = {
    OR = {
        AND = {
            NOT = { OR = {
                has_game_rule = { rule = pwc_protection_period option = DAYS_90 }
                has_game_rule = { rule = pwc_protection_period option = NONE }
            } }
            date < 2026.6.30
        }
        AND = {
            has_game_rule = { rule = pwc_protection_period option = DAYS_90 }
            date < 2026.4.1
        }
    }
}


pwc_st_ai_war_pacing_ready = {
    OR = {
        is_ai = no
        AND = {
            NOT = { has_country_flag = pwc_flag_coalition_reserved }
            NOT = { AND = {
                has_variable = pwc_coalition_target
                has_war_with = var:pwc_coalition_target
            } }
        }
    }
    OR = {
        is_ai = no
        has_game_rule = { rule = pwc_ai_war_pacing option = OFF }
        AND = {
            has_country_flag = pwc_flag_war_phase_armed
            NOT = { has_country_flag = pwc_flag_war_phase_pending }
            NOT = { has_country_flag = pwc_flag_war_cooldown }
        }
    }
}
