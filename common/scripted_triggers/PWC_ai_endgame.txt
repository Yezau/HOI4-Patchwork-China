pwc_st_coalition_player = {
    exists = yes
    is_ai = no
    pwc_st_is_city_country = yes
    pwc_st_city_alive = yes
    is_subject = no
    has_capitulated = no
    NOT = { is_in_peace_conference = yes }
    pwc_st_protection_active = no
}


pwc_st_coalition_partner = {
    exists = yes
    is_ai = yes
    pwc_st_is_city_country = yes
    pwc_st_city_alive = yes
    is_subject = no
    is_in_faction = no
    has_capitulated = no
    NOT = { is_in_peace_conference = yes }
    num_divisions > 2
    is_neighbor_of = event_target:pwc_coalition_player
    NOT = { has_non_aggression_pact_with = event_target:pwc_coalition_player }
    NOT = { has_truce_with = event_target:pwc_coalition_player }
    NOT = { has_guaranteed = event_target:pwc_coalition_player }
    NOT = { has_military_access_to = event_target:pwc_coalition_player }
    event_target:pwc_coalition_player = {
        NOT = { has_guaranteed = PREV }
        NOT = { has_military_access_to = PREV }
        NOT = { is_in_faction_with = PREV }
    }
    
    
    NOT = { any_enemy_country = {
        NOT = { tag = event_target:pwc_coalition_player }
    } }
}

pwc_st_coalition_size_ready = {
    check_variable = {
        var = pwc_coalition_count value = 2 compare = greater_than_or_equals
    }
    check_variable = {
        var = pwc_coalition_manpower value = pwc_coalition_required_manpower
        compare = greater_than_or_equals
    }
    check_variable = {
        var = pwc_coalition_industry value = pwc_coalition_required_industry
        compare = greater_than_or_equals
    }
}

pwc_st_ai_only_war = {
    exists = yes
    is_ai = yes
    pwc_st_is_city_country = yes
    pwc_st_city_alive = yes
    has_capitulated = no
    has_war = yes
    NOT = { any_enemy_country = { is_ai = no } }
}

pwc_st_ai_finish_war = {
    pwc_st_ai_only_war = yes
    has_country_flag = pwc_flag_ai_war_tracked
    NOT = { has_country_flag = pwc_flag_ai_war_grace }
    enemies_strength_ratio > 1.25
}
