defined_text = {
    name = PWCGetUnitCapacityCount
    text = {
        trigger = { NOT = { pwc_st_unit_limit_applies = yes } }
        localization_key = PWC_UI_UNIT_CAPACITY_COUNT_DISABLED
    }
    text = {
        trigger = { check_variable = { num_divisions > pwc_unit_capacity } }
        localization_key = PWC_UI_UNIT_CAPACITY_COUNT_ABOVE
    }
    text = {
        trigger = { check_variable = { num_divisions = pwc_unit_capacity } }
        localization_key = PWC_UI_UNIT_CAPACITY_COUNT_EQUAL
    }
    text = {
        trigger = { always = yes }
        localization_key = PWC_UI_UNIT_CAPACITY_COUNT_BELOW
    }
}
