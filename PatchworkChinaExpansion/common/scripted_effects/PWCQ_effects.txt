pwcq_init = {
    set_global_flag = pwcq_started
    pwch_ensure_counted = yes
    every_country = {
        limit = { pwc_st_live_city_country = yes NOT = { has_country_flag = pwcq_setup } }
        pwcq_setup = yes
    }
}
pwcq_setup = {
    set_country_flag = pwcq_setup
    if = {
        limit = { capital_scope = { is_coastal = yes } }
        set_country_flag = pwcq_coastal
    }
    set_variable = { pwcq_stage = 1 }
    if = {
        limit = { check_variable = { pwch_states > 5 } }
        set_variable = { pwcq_stage = 2 }
    }
    if = {
        limit = { check_variable = { pwch_states > 14 } }
        set_variable = { pwcq_stage = 3 }
    }
    set_variable = { pwcq_preset = 1 }
    if = {
        limit = { is_ai = yes }
        if = {
            limit = { has_country_flag = pwc_ai_mobility }
            set_variable = { pwcq_preset = 3 }
        }
        else_if = {
            limit = { has_country_flag = pwc_ai_air_power }
            set_variable = { pwcq_preset = 3 }
        }
        else_if = {
            limit = { has_country_flag = pwc_ai_industrial }
            set_variable = { pwcq_preset = 2 }
        }
        else_if = {
            limit = { has_country_flag = pwc_ai_defense }
            set_variable = { pwcq_preset = 4 }
        }
    }
    pwcq_apply_preset = yes
    pwcq_recount = yes
}
