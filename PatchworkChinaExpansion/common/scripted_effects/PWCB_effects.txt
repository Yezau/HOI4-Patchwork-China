pwcb_reset_endurance = {
    set_variable = { pwcb_hold = 0 }
    set_variable = { pwcb_last_stand = 0 }
    clr_country_flag = pwcb_holding
    clr_country_flag = pwcb_last_standing
}
pwcb_sync_war_foes = {
    clear_array = pwcb_war_foes
    if = {
        limit = { has_war = no }
        pwcb_reset_endurance = yes
    }
    every_enemy_country = {
        add_to_array = { array = PREV.pwcb_war_foes value = THIS }
    }
}
pwcb_init_war_foes = {
    set_global_flag = pwcb_war_roster_v1
    every_country = {
        limit = { pwc_st_live_city_country = yes has_war = yes }
        if = {
            limit = { NOT = { has_country_flag = pwcb_in_war } }
            set_country_flag = pwcb_in_war
        }
        pwcb_sync_war_foes = yes
    }
}
pwcb_refresh = {
    if = {
        limit = { NOT = { has_dynamic_modifier = { modifier = pwcb_art_of_war } } }
        add_dynamic_modifier = { modifier = pwcb_art_of_war }
    }
    force_update_dynamic_modifier = yes
}
pwcb_learn_shiji = {
    if = {
        limit = { NOT = { has_country_flag = pwcb_ch_shiji } }
        set_country_flag = pwcb_ch_shiji
        add_to_variable = { pwcb_learnt = 1 }
        add_to_variable = { pwcb_s_planning_speed = 0.1 }
        set_country_flag = pwcd_news
        if = {
            limit = { check_variable = { pwcb_learnt > 12 } }
            set_country_flag = pwcb_sage
            add_to_variable = { pwcb_s_experience_gain_army_factor = 0.1 }
            add_to_variable = { pwcb_s_army_org_factor = 0.05 }
        }
        pwcb_refresh = yes
    }
}

pwcb_check_endurance = {
    if = {
        limit = { NOT = { has_country_flag = pwcb_ch_junxing } }
        if = {
            limit = { has_war = yes enemies_strength_ratio > 1 }
            if = {
                limit = { NOT = { has_country_flag = pwcb_holding } }
                set_country_flag = pwcb_holding
                set_variable = { pwcb_hold = 0 }
            }
            if = {
                limit = { has_country_flag = { flag = pwcb_holding days > 29 } }
                set_variable = { pwcb_hold = 1 }
            }
            if = {
                limit = { has_country_flag = { flag = pwcb_holding days > 59 } }
                set_variable = { pwcb_hold = 2 }
            }
            if = {
                limit = { has_country_flag = { flag = pwcb_holding days > 89 } }
                set_variable = { pwcb_hold = 3 }
            }
            if = {
                limit = { check_variable = { pwcb_hold > 2 } }
                pwcb_learn_junxing = yes
            }
        }
        else = {
            clr_country_flag = pwcb_holding
            set_variable = { pwcb_hold = 0 }
        }
    }
    if = {
        limit = { NOT = { has_country_flag = pwcb_ch_jiudi } }
        if = {
            limit = { has_war = yes check_variable = { var = enemies_strength_ratio value = 3 compare = greater_than_or_equals } }
            if = {
                limit = { NOT = { has_country_flag = pwcb_last_standing } }
                set_country_flag = pwcb_last_standing
                set_variable = { pwcb_last_stand = 0 }
            }
            if = {
                limit = { has_country_flag = { flag = pwcb_last_standing days > 29 } }
                set_variable = { pwcb_last_stand = 1 }
            }
            if = {
                limit = { has_country_flag = { flag = pwcb_last_standing days > 59 } }
                set_variable = { pwcb_last_stand = 2 }
            }
            if = {
                limit = { check_variable = { pwcb_last_stand > 1 } }
                pwcb_learn_jiudi = yes
            }
        }
        else = {
            clr_country_flag = pwcb_last_standing
            set_variable = { pwcb_last_stand = 0 }
        }
    }
}
pwcb_daily = {
    every_country = {
        limit = { pwc_st_live_city_country = yes OR = { has_war = yes has_country_flag = pwcb_holding has_country_flag = pwcb_last_standing } OR = { NOT = { has_country_flag = pwcb_ch_junxing } NOT = { has_country_flag = pwcb_ch_jiudi } } }
        pwcb_check_endurance = yes
    }
}
pwcb_monthly = {
    every_country = {
        limit = { pwc_st_live_city_country = yes }
        set_variable = { pwcb_mountains = 0 }
        every_controlled_state = {
            limit = { has_state_flag = pwcb_mountain }
            PREV = { add_to_variable = { pwcb_mountains = 1 } }
        }
        if = {
            limit = { check_variable = { pwcb_mountains > 2 } }
            pwcb_learn_dixing = yes
        }
        pwcb_check_endurance = yes
        if = {
            limit = { num_of_controlled_states > 9 }
            pwcb_learn_xingjun = yes
        }
        if = {
            limit = { check_variable = { num_deployed_planes > 49 } }
            pwcb_learn_huogong = yes
        }
        if = {
            limit = { check_variable = { enemies^num > 2 } }
            pwcb_learn_jiubian = yes
        }
        set_variable = { pwcb_month_taken = 0 }
        clear_array = pwcb_taken_states
    }
}
pwcb_weekly = {
    every_country = {
        limit = { pwc_st_live_city_country = yes }
        if = {
            limit = { has_war = yes }
            add_to_variable = { pwcb_war_weeks = 1 }
        }
        else = {
            clr_country_flag = pwcb_in_war
            clear_array = pwcb_war_foes
            pwcb_reset_endurance = yes
            set_variable = { pwcb_war_weeks = 0 }
        }
    }
}
