pwco_init = {
    set_global_flag = pwco_started
    set_variable = { global.pwco_debris = 0 }
    pwch_ensure_counted = yes
    every_country = {
        limit = { pwc_st_live_city_country = yes }
        pwco_on_count = yes
    }
}
pwco_show_report = {
    if = {
        limit = {
            NOT = { has_country_flag = pwco_report_open }
            NOT = { has_country_flag = pwco_report_wait }
            check_variable = { pwco_reports_kind^num > 0 }
        }
        set_variable = { pwco_report_kind = pwco_reports_kind^0 }
        set_variable = { pwco_report_by = pwco_reports_by^0 }
        remove_from_array = { array = pwco_reports_kind index = 0 }
        remove_from_array = { array = pwco_reports_by index = 0 }
        set_country_flag = pwco_report_open
        if = {
            limit = { check_variable = { pwco_report_kind = 1 } }
            country_event = { id = pwco.1 }
        }
        else_if = {
            limit = { check_variable = { pwco_report_kind = 2 } }
            country_event = { id = pwco.2 }
        }
        else_if = {
            limit = { check_variable = { pwco_report_kind = 3 } }
            country_event = { id = pwco.3 }
        }
    }
}
pwco_close_report = {
    clr_country_flag = pwco_report_open
    set_country_flag = pwco_report_wait
    country_event = { id = pwco.6 hours = 1 }
}
pwco_on_count = {
    set_variable = { pwco_site = 0 }
    if = {
        limit = { owns_state = 387 controls_state = 387 }
        set_variable = { pwco_site = 4 }
    }
    if = {
        limit = { owns_state = 277 controls_state = 277 }
        set_variable = { pwco_site = 3 }
    }
    if = {
        limit = { owns_state = 87 controls_state = 87 }
        set_variable = { pwco_site = 2 }
    }
    if = {
        limit = { owns_state = 56 controls_state = 56 }
        set_variable = { pwco_site = 1 }
    }
    set_variable = { pwco_access = 0 }
    if = {
        limit = { check_variable = { pwch_states > 29 } }
        set_variable = { pwco_access = 2 }
    }
    else_if = {
        limit = { check_variable = { pwch_states > 14 } check_variable = { pwco_site > 0 } }
        set_variable = { pwco_access = 1 }
    }
    set_variable = { pwco_slots = 0 }
    if = {
        limit = { check_variable = { pwco_access > 0 } }
        set_variable = { pwco_slots = 1 }
        if = {
            limit = { check_variable = { pwch_states > 29 } }
            set_variable = { pwco_slots = 2 }
        }
        if = {
            limit = { check_variable = { pwch_states > 49 } }
            set_variable = { pwco_slots = 3 }
        }
    }
    if = {
        limit = { check_variable = { pwco_access > 0 } NOT = { has_country_flag = pwco_told } }
        set_country_flag = pwco_told
        if = {
            limit = { is_ai = no }
            country_event = { id = pwco.5 }
            set_country_flag = pwch_news
        }
    }
}
